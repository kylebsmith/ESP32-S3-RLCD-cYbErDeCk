-- in
  >toggle hat clap bass

-- hook
  >toggle lead

  >toggle pad pad:2 bass bass:2 lead arp
  >route box:x bass:2
  >route box:y arp
  >route ramp pad:2
  >send view lines 9

  >toggle pad pad:2 bass bass:2 lead arp
  >route box:x bass
  >route box:y lead
  >route ramp pad
  >send view lines 3
